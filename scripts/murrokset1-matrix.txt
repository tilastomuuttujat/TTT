function murrosMatrixRoute(data,mode,start){
 const candidates=data.edges.filter(edge=>edge.modes.includes(mode)),nodes=new Set([start]),edges=new Set();
 let changed=true;while(changed){changed=false;for(const edge of candidates)if(nodes.has(edge.source)&&!edges.has(edge.id)){edges.add(edge.id);nodes.add(edge.target);changed=true;}}
 return {nodes,edges,candidates};
}
function murrosMatrixLayout(data,expanded=new Set()){
 const positions=new Map(),visible=new Set();
 for(const n of data.nodes){if(n.kind==='anchor'){positions.set(n.id,{x:225+((n.rank??n.chapter)%4)*430,y:240+Math.floor((n.rank??n.chapter)/4)*420});visible.add(n.id);}}
 for(const n of data.nodes.filter(n=>n.kind==='petal')){const c=positions.get(n.parent),angle=-Math.PI/2+n.order*Math.PI*2/data.nodes.filter(v=>v.parent===n.parent&&v.kind==='petal').length;positions.set(n.id,{x:c.x+Math.cos(angle)*110,y:c.y+Math.sin(angle)*110});visible.add(n.id);}
 for(const n of data.nodes.filter(n=>n.kind==='mechanism')){const p=positions.get(n.parent),petal=data.nodes.find(v=>v.id===n.parent),angle=-Math.PI/2+petal.order*Math.PI*2/data.nodes.filter(v=>v.parent===petal.parent&&v.kind==='petal').length;const siblings=data.nodes.filter(v=>v.parent===n.parent),i=siblings.findIndex(v=>v.id===n.id),spread=(i-(siblings.length-1)/2)*40;positions.set(n.id,{x:p.x+Math.cos(angle)*85-Math.sin(angle)*spread,y:p.y+Math.sin(angle)*85+Math.cos(angle)*spread});if(expanded.has(n.parent))visible.add(n.id);}
 const path=edge=>{const a=positions.get(edge.source),b=positions.get(edge.target),dx=b.x-a.x,dy=b.y-a.y;return `M${a.x},${a.y} C${a.x+dx*.35-dy*.15},${a.y+dy*.35+dx*.15} ${a.x+dx*.65-dy*.15},${a.y+dy*.65+dx*.15} ${b.x},${b.y}`;};
 return {positions,visible,path,width:1780,height:Math.max(900,Math.ceil(data.nodes.filter(n=>n.kind==='anchor').length/4)*420+60)};
}
function murrosProjectEdges(data,edges,visible){
 const byId=new Map(data.nodes.map(n=>[n.id,n])),groups=new Map();
 const project=id=>{let n=byId.get(id);while(n&&!visible.has(n.id))n=byId.get(n.parent);return n?.id;};
 for(const edge of edges){if(edge.kind==='membership')continue;const source=project(edge.source),target=project(edge.target);if(!source||!target||source===target)continue;const key=source+'>'+target+':'+edge.status;if(!groups.has(key))groups.set(key,{...edge,id:key,source,target,members:[]});groups.get(key).members.push(edge);}
 return [...groups.values()];
}
function murrosFlowerPartners(data,id,mode){
 const byId=new Map(data.nodes.map(n=>[n.id,n])),seed=byId.get(id);if(!seed)return [];
 const descendants=new Set([id]);let changed=true;while(changed){changed=false;for(const n of data.nodes)if(n.parent&&descendants.has(n.parent)&&!descendants.has(n.id)){descendants.add(n.id);changed=true;}}
 const groups=new Map();for(const edge of data.edges){if(edge.kind==='membership'||!edge.modes.includes(mode))continue;const a=descendants.has(edge.source),b=descendants.has(edge.target);if(a===b)continue;let other=byId.get(a?edge.target:edge.source);while(other?.kind==='mechanism')other=byId.get(other.parent);if(!other||other.chapter===seed.chapter)continue;if(!groups.has(other.id))groups.set(other.id,{node:other,edges:[]});groups.get(other.id).edges.push(edge);}return [...groups.values()];
}
function murrosLabelLines(label){const lines=[''];for(const word of label.split(' ')){if((lines.at(-1)+' '+word).trim().length>19&&lines.at(-1))lines.push(word);else lines[lines.length-1]=(lines.at(-1)+' '+word).trim();}return lines;}
