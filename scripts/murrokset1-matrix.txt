function murrosMatrixRoute(data,mode,start){
 const candidates=data.edges.filter(edge=>edge.modes.includes(mode)),nodes=new Set([start]),edges=new Set();
 let changed=true;while(changed){changed=false;for(const edge of candidates)if(nodes.has(edge.source)&&!edges.has(edge.id)){edges.add(edge.id);nodes.add(edge.target);changed=true;}}
 return {nodes,edges,candidates};
}
function murrosMatrixLayout(data){
 const positions=new Map(data.nodes.map(node=>[node.id,{x:220+node.chapter*260,y:85+node.level*140+(node.offset||0)*36}]));
 const path=edge=>{const a=positions.get(edge.source),b=positions.get(edge.target);if(a.x===b.x)return `M${a.x},${a.y} C${a.x+65},${a.y+45} ${b.x+65},${b.y-45} ${b.x},${b.y}`;return `M${a.x},${a.y} C${a.x+110},${a.y} ${b.x-110},${b.y} ${b.x},${b.y}`;};
 return {positions,path};
}
