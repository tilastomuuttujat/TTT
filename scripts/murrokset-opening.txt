function Cf({cases,selectedId,onSelect,lens,relations=[],focus}){
 const evidenceRank={measured:3,sourced:2,logic:1,hypothesis:0};
 const active=cases.filter(c=>(lens.theme==='all'||vn(c).includes(lens.theme))&&(lens.evidence==='all'||evidenceRank[yn(c).level]>=evidenceRank[lens.evidence]));
 const featured=active.filter(c=>Number.isFinite(c.data?.overview?.featured_order)).sort((a,b)=>a.data.overview.featured_order-b.data.overview.featured_order);
 const choices=featured.length?featured:active;
 const [chosen,setChosen]=S.useState(''),[phase,setPhase]=S.useState(0),[playing,setPlaying]=S.useState(()=>!window.matchMedia?.('(prefers-reduced-motion: reduce)').matches);
 const current=choices.find(c=>c.id===chosen)||choices[0],model=current?openingModel(current):null;
 const nodes=S.useRef([]);
 S.useEffect(()=>{setPhase(0);},[current?.id]);
 S.useEffect(()=>{const media=window.matchMedia?.('(prefers-reduced-motion: reduce)');const change=()=>{if(media?.matches)setPlaying(false);};media?.addEventListener?.('change',change);return()=>media?.removeEventListener?.('change',change);},[]);
 S.useEffect(()=>{if(!playing||!current||selectedId||document.hidden)return;if(phase>=2){setPlaying(false);return;}const timer=window.setTimeout(()=>setPhase(p=>p+1),6500);return()=>window.clearTimeout(timer);},[playing,phase,current?.id,selectedId]);
 const select=c=>{setChosen(c.id);setPhase(0);setPlaying(false);};
 const fields=[['production','Tuotanto'],['livelihood','Toimeentulo'],['responsibility','Yhteinen vastuu']];
 const steps=['1. Mitä tuotannossa muuttuu?','2. Kenelle muutos välittyy?','3. Kuka vastaa yhteensovittamisesta?'];
 const patterns={converging:[105,140,175],diverging:[45,140,235],unclear:[70,140,210]};
 const ends=patterns[model?.pattern]||patterns.unclear;
 return L.jsxs('section',{className:'hypothesis-opening',children:[
  L.jsx('p',{className:'kicker',children:'Iso kuva · tutkimuksen mukana tarkentuva tulkinta'}),
  L.jsx('h1',{children:'Miten tuotannon muutos välittyy ihmisten toimeentuloon ja yhteiseen vastuuseen?'}),
  L.jsx('p',{className:'opening-lead',children:'Ajankohdan asetelma näkyy jo painikkeessa. Vertaa, muuttuivatko tuotanto, toimeentulo ja yhteinen vastuu samaan tahtiin — vai jäikö niiden välille kuilu.'}),
  L.jsx('nav',{'aria-label':'Ajankohtien asetelmat',className:'era-arrangements',children:choices.map((c,i)=>{const m=openingModel(c);return L.jsxs('button',{type:'button',className:`era-arrangement ${c.id===current?.id?'active':''}`,'aria-pressed':c.id===current?.id,ref:el=>nodes.current[i]=el,onClick:()=>select(c),onKeyDown:event=>{if(['ArrowRight','ArrowLeft','Home','End'].includes(event.key)){event.preventDefault();const j=event.key==='Home'?0:event.key==='End'?choices.length-1:(i+(event.key==='ArrowRight'?1:-1)+choices.length)%choices.length;select(choices[j]);nodes.current[j]?.focus();}},children:[L.jsx('span',{className:'era-period',children:m.period}),L.jsx('strong',{children:m.title}),L.jsx('span',{className:'era-vertical',children:fields.map(([key,label])=>L.jsxs('span',{className:`arrangement-row row-${key}`,children:[L.jsx('small',{children:label}),L.jsx('span',{children:m[key]})]},key))}),L.jsx('small',{className:`research-state state-${m.state}`,children:m.stateLabel})]},c.id);})}),
  !model?L.jsx('p',{children:'Rajauksessa ei ole ankkureita. Laajenna rajausta nähdäksesi tutkittavat asetelmat.'}):L.jsxs('div',{className:'opening-story',children:[
   L.jsxs('div',{className:'story-heading',children:[L.jsx('span',{className:`research-state state-${model.state}`,children:model.stateLabel}),L.jsx('span',{children:model.period}),L.jsx('h2',{children:model.hypothesis})]}),
   L.jsxs('div',{className:'guided-arrangement',children:[L.jsxs('div',{className:`arrangement-sketch phase-${phase} ${playing?'is-playing':''}`,children:[L.jsxs('svg',{viewBox:'0 0 640 280',role:'img','aria-label':`Käsitteellinen luonnos: ${model.patternLabel}. Ei tilastollinen asteikko.`,children:[L.jsx('defs',{children:L.jsxs('linearGradient',{id:'opening-wash',children:[L.jsx('stop',{offset:'0%',stopColor:'#e3efe8'}),L.jsx('stop',{offset:'100%',stopColor:'#faf3e5'})]})}),L.jsx('rect',{x:0,y:0,width:640,height:280,fill:'url(#opening-wash)',rx:20}),...fields.map(([key,label],i)=>L.jsxs('g',{className:`sketch-lane lane-${i} ${i<=phase?'revealed':''}`,children:[L.jsx('path',{className:`sketch-path path-${key}`,d:`M30 ${70+i*70} C230 ${70+i*70},350 ${ends[i]},600 ${ends[i]}`,fill:'none',strokeWidth:7}),L.jsx('circle',{cx:600,cy:ends[i],r:8}),L.jsx('text',{x:30,y:52+i*70,children:label})]},key))]}),L.jsx('p',{className:'sketch-caption',children:`${model.patternLabel}. Viivojen lähentyminen tai erkaneminen kuvaa tutkittavaa ajatusta, ei mitattua muutosta tai vaikutuksen suuruutta.`})]}),L.jsxs('div',{className:'guided-explanation',children:[L.jsx('ol',{className:'guide-steps',children:fields.map(([key],i)=>L.jsxs('li',{className:i===phase?'active':i<phase?'done':'pending',children:[L.jsx('button',{type:'button','aria-current':i===phase?'step':undefined,onClick:()=>{setPhase(i);setPlaying(false);},children:steps[i]}),L.jsx('p',{children:model[key]})]},key))}),L.jsxs('div',{className:'guide-controls',children:[L.jsx('button',{type:'button',onClick:()=>{if(playing)setPlaying(false);else{if(phase===2)setPhase(0);setPlaying(true);}},children:playing?'Pysäytä opastus':phase===2?'Näytä opastus uudelleen':'Näytä ajatus vaiheittain'}),L.jsx('span',{'aria-live':'polite',children:`Vaihe ${phase+1}/3`})]})]})]}),
   L.jsxs('div',{className:'research-summary',children:[L.jsx('p',{children:`Ajanjakson kysymystä avaa ankkuri: ${current.title} (${current.year_start}–${current.year_end}). Se ei yksin edusta koko aikakauden tai kaikkien ryhmien kokemusta.`}),L.jsxs('p',{children:[L.jsx('strong',{children:'Mitä tästä tutkitaan? '}),model.question]}),L.jsx('p',{children:'Asetelma on tulkinta. Tilastohavainto ja sitä selittävä mekanismi arvioidaan erikseen; lisääntyvä aineisto ei yksin vahvista syy-yhteyttä.'}),L.jsx(on,{variant:'outline',onClick:()=>{setPlaying(false);onSelect(current.id,50);},children:`Tutki ankkuria: ${current.title}`})]}),
   L.jsxs('details',{className:'research-evidence',children:[L.jsx('summary',{children:'Mihin tulkinta nojaa — ja mikä voisi muuttaa sitä?'}),L.jsx('dl',{children:[['Tukeva näyttö',model.support],['Vastanäyttö',model.counter],['Vaihtoehtoinen selitys',model.alternative],['Seuraava tutkimuskysymys',model.question],['Tulkinnan tarkistus',model.reviewed||'Tarkistuspäivää ei ole kirjattu.']].map(([label,value])=>L.jsxs('div',{children:[L.jsx('dt',{children:label}),L.jsx('dd',{children:value})]},label))})]})
  ]}),
  L.jsxs('details',{className:'opening-anchors',children:[L.jsx('summary',{children:`Syvenny muihin ankkureihin (${active.length})`}),L.jsx('div',{className:'anchor-arrangements',children:active.map(c=>L.jsxs('button',{type:'button',onClick:()=>{setPlaying(false);onSelect(c.id,50);},className:`anchor-arrangement ${selectedId===c.id?'selected':''} ${focus&&c.year_start>=focus[0]&&c.year_start<focus[1]?'tour-focus':''}`,children:[L.jsx('small',{children:`${c.year_start}–${c.year_end}`}),L.jsx('strong',{children:c.title}),L.jsx('span',{children:c.data.summary||'Asetelmaa ei ole vielä kuvattu.'}),L.jsx('small',{children:openingModel(c).stateLabel})]},c.id))})]}),
  L.jsxs('details',{className:'opening-relations',children:[L.jsx('summary',{children:'Täydentävä näkymä: ankkurien väliset kytkennät'}),L.jsx(RelationView,{cases,selectedId,onSelect,lens,relations})]})
 ]});
}
