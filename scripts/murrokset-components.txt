function Sf({series:e,values:t}){
 const groups=[...new Map(t.filter(p=>p.series_key===e.series_key).map(p=>[p.group_key,p.group_label||p.group_key])).entries()];const [selected,setGroup]=S.useState('FI');const group=groups.some(([key])=>key===selected)?selected:groups[0]?.[0];const model=chartModel(e,t,group);
 if(!model)return L.jsx('p',{children:'Tällä sarjalla ja ryhmällä ei ole julkaistuja arvoja.'});
 return L.jsxs('figure',{className:'atlas-chart',children:[L.jsxs('figcaption',{children:[L.jsx('span',{children:e.title}),L.jsx('small',{children:e.unit})]}),L.jsxs('label',{children:['Aineiston ryhmä ',L.jsx('select',{value:group,onChange:event=>setGroup(event.target.value),children:groups.map(([key,label])=>L.jsx('option',{value:key,children:label},key))})]}),L.jsxs('svg',{viewBox:'0 0 580 195',role:'img','aria-label':`${e.title}, ${group}, ${model.minYear}–${model.maxYear}`,children:[L.jsx('path',{className:'chart-grid',d:'M60 45H530M60 160H530'}),...model.segments.map((segment,i)=>L.jsx('path',{className:segment.a.value_kind==='observed'?'chart-line':'chart-line chart-other',d:segment.path},i)),...model.rows.map((p,i)=>L.jsxs('g',{children:[L.jsx('circle',{cx:model.x(p.year),cy:model.y(+p.value),r:3,fill:p.value_kind==='observed'?'#176b5b':'#9a641d'}),L.jsx('title',{children:`${p.year}: ${p.value} · ${p.value_kind||'tietotila puuttuu'}`})]},i)),L.jsx('text',{x:55,y:49,textAnchor:'end',children:model.hi.toLocaleString('fi-FI',{maximumFractionDigits:2})}),L.jsx('text',{x:55,y:164,textAnchor:'end',children:model.lo.toLocaleString('fi-FI',{maximumFractionDigits:2})}),L.jsx('text',{x:60,y:185,children:model.minYear}),L.jsx('text',{x:530,y:185,textAnchor:'end',children:model.maxYear})]}),L.jsx('p',{children:'Vihreä: havaittu arvo.
