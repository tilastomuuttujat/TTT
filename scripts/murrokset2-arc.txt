function TemporalTheme({model,step,onStep}){
 const v=model.visualization,years=v.years,ys=[45,112,179],x=year=>155+(year-years[0])/(years[3]-years[0])*775;
 const stage=model.steps[step].period;
 const activeStart=x(stage.start),activeEnd=x(stage.end);
 C.useEffect(()=>{const current=document.querySelector(`[data-time-model="${model.id}"] .time-scroll`);if(current)current.scrollTo({left:Math.max(0,activeStart-current.clientWidth*.3),behavior:window.matchMedia('(prefers-reduced-motion: reduce)').matches?'instant':'smooth'});},[step,model.id]);
 return I.jsxs('div',{className:`time-model time-model-${v.kind}`,'data-time-model':model.id,children:[
 I.jsxs('div',{className:'time-model-heading',children:[I.jsx('strong',{children:v.title}),I.jsx('span',{children:`${stage.start}–${stage.end} · ${step===2?'ehdollinen tulevaisuus':'historiallisen jakson hypoteesi'}`})]}),
 I.jsx('div',{className:'time-scroll',tabIndex:0,'aria-label':'Ajallinen vaikutusmalli. Vaakasuunta on kalenteriaika; rivit kuvaavat vaikutusreittejä, eivät suureita.',children:I.jsxs('svg',{viewBox:'0 0 1000 285',role:'img','aria-label':`${v.title}. ${v.lanes.join(', ')}. Aktiivinen jakso ${stage.start}–${stage.end}. Reitit ovat hypoteeseja, leveys ei kuvaa määriä.`,children:[
 I.jsx('rect',{className:'time-future-band',x:x(2026),y:12,width:x(years[3])-x(2026),height:225}),
 I.jsx('rect',{className:'time-active-band',x:activeStart,y:12,width:activeEnd-activeStart,height:225}),
 ...years.map((year,i)=>I.jsxs('g',{children:[I.jsx('line',{className:'time-grid',x1:x(year),x2:x(year),y1:15,y2:239}),I.jsx('text',{className:'time-year',x:x(year),y:260,textAnchor:'middle',children:year})]},year)),
 ...v.lanes.map((label,row)=>I.jsxs('g',{children:[I.jsx('text',{className:'time-lane-label',x:6,y:ys[row]+4,children:label}),I.jsx('line',{className:'time-lane',x1:155,x2:930,y1:ys[row],y2:ys[row]})]},row)),
 ...[0,1,2].flatMap(phase=>v.stage_routes[phase].map(([from,to],id)=>{
 const a=x(years[phase]),b=x(years[phase+1]),mid=(a+b)/2;
 const path=`M${a} ${ys[from]} C${mid} ${ys[from]} ${mid} ${ys[to]} ${b} ${ys[to]}`;
 return I.jsxs('g',{className:`time-route stage-${phase} ${phase===step?'is-current':''} ${phase>step?'is-pending':''} ${phase===2?'is-scenario':''}`,'data-stage':phase,children:[I.jsx('path',{className:`time-route-line route-lane-${to}`,d:path,pathLength:1}),phase===step&&I.jsx('path',{className:`time-traveller route-lane-${to}`,d:path,pathLength:1}),I.jsx('circle',{className:`time-arrival route-lane-${to}`,cx:b,cy:ys[to],r:3})]},`${phase}-${id}`);
 })),
 ...[0,1,2].map(phase=>I.jsx('g',{className:`time-period ${phase===step?'is-current':''}`,role:'button',tabIndex:0,'aria-label':`Toista ${years[phase]}–${years[phase+1]}`,onClick:()=>onStep(phase),onKeyDown:event=>{if(event.key==='Enter'||event.key===' '){event.preventDefault();onStep(phase);}},children:I.jsx('rect',{x:x(years[phase]),y:269,width:x(years[phase+1])-x(years[phase]),height:6,rx:3})},phase))
 ]})}),
 I.jsx('p',{className:'time-model-legend',children:'Aika → · Rivit = vaikutusreittejä · Virtaus = oletettu yhteys, ei määrä · Katkoviiva = ehdollinen tulevaisuus'}),
 I.jsx('p',{className:'time-period-basis',children:v.period_basis})
 ]});
}
function Ln({step,onStep,modelId,onPause}){
 const model=themeModels.models.find(m=>m.id===modelId)||themeModels.models[0];
 const statusLabel={untested:'Validius arvioimatta',provisional:'Alustavaa näyttöä liitetty',supported:'Näyttöä määritellyille yhteyksille',challenged:'Näyttö haastaa mallin'}[model.status];
 const [open,setOpen]=C.useState(false),dialog=C.useRef(null),trigger=C.useRef(null);
 const close=()=>{setOpen(false);window.requestAnimationFrame(()=>trigger.current?.focus());};
 C.useLayoutEffect(()=>{if(!open)return;const node=dialog.current;node.showModal();const previous=document.body.style.overflow;document.body.style.overflow='hidden';return()=>{node.close();document.body.style.overflow=previous;};},[open]);
 return I.jsxs('div',{className:`impact-arc thematic-arc temporal-arc impact-step-${step}`,'aria-label':`${model.title}. Tutkittava hypoteesi: ${model.hypothesis}`,children:[
 I.jsx('p',{className:'impact-note',children:`Hypoteesimalli · ${statusLabel.toLowerCase()} · reitit eivät ole mittauksia; ajanjaksot ovat tutkittavia rajauksia`}),
 I.jsx('button',{ref:trigger,type:'button',className:'theme-probe',onClick:()=>{onPause();setOpen(true);},children:'Koettele hypoteesia'}),
 I.jsx(TemporalTheme,{model,step,onStep}),
 open&&I.jsxs('dialog',{ref:dialog,className:'theme-research-dialog','aria-labelledby':'theme-research-title',onCancel:event=>{event.preventDefault();close();},children:[
 I.jsx('button',{type:'button',className:'theme-research-close','aria-label':'Sulje hypoteesin tarkastelu',onClick:close,autoFocus:true,children:'×'}),
 I.jsxs('article',{children:[I.jsx('small',{children:statusLabel+' · versio '+model.revision}),I.jsx('h1',{id:'theme-research-title',children:model.title+' — '+model.angle}),I.jsx('h2',{children:'Hypoteesi'}),I.jsx('p',{children:model.hypothesis}),I.jsx('h2',{children:'Kilpaileva selitys'}),I.jsx('p',{children:model.alternative}),I.jsx('h2',{children:'Mallin oletukset'}),I.jsx('ul',{children:model.assumptions.map((x,i)=>I.jsx('li',{children:x},i))}),I.jsx('h2',{children:'Mitä pitäisi havainnoida?'}),I.jsx('ul',{children:model.observe.map((x,i)=>I.jsx('li',{children:x},i))}),I.jsx('h2',{children:'Mikä heikentäisi tai kumoaisi tulkintaa?'}),I.jsx('p',{children:model.refutation}),I.jsx('h2',{children:'Tutkimusnäytön tila'}),I.jsx('p',{children:(model.status==='untested'?'Tämän mallin validiutta ei ole arvioitu. ':'Liitetty näyttö koskee nimettyjä yhteyksiä ja rajattuja olosuhteita. ')+ 'Visualisoinnin reitit ja värit jäsentävät mahdollista mekanismia; ne eivät ole historiallisia havaintoja, mitattuja vaikutuksia tai ennusteita. Tuleva näyttö voi vahvistaa osan yhteyksistä, rajata soveltamisalaa tai kumota mallin.'}),...model.evidence.map((e,i)=>I.jsxs('section',{className:'theme-evidence-entry',children:[I.jsx('a',{href:e.url,target:'_blank',rel:'noopener noreferrer',children:e.title}),I.jsx('p',{children:e.claim})]},i)),I.jsx('p',{children:'Ankkuritapauksilla ja niiden kertomuksilla on oma lähdepohjansa. Se ei automaattisesti validoi tätä laajempaa teemamallia.'})]})]})]});
}
