function kn({index,playing,onIndex,onToggle,onClose}){
 const chapter=Tn[index],phases=chapter.phases||[{title:'Katso ajanjakson asetelmaa',text:chapter.text},{title:'Tutki yhteyksiä',text:'Viivan suunta on havainnollistava tulkinta, ei mitattu historiallinen vaikutus.'},{title:'Koettele tulkintaa',text:'Avaa ajanjakson ankkurit ja vertaa havaintoja, lähteitä sekä vaihtoehtoisia selityksiä.'}];
 const [phase,setPhase]=S.useState(0);
 S.useEffect(()=>setPhase(0),[index]);
 S.useEffect(()=>{if(!playing||document.hidden)return;const timer=window.setTimeout(()=>{if(phase<phases.length-1)setPhase(p=>p+1);else if(index<Tn.length-1)onIndex(index+1);else onToggle();},9000);return()=>window.clearTimeout(timer);},[playing,index,phase]);
 S.useEffect(()=>{const media=window.matchMedia?.('(prefers-reduced-motion: reduce)');const stop=()=>{if(media?.matches&&playing)onToggle();};media?.addEventListener?.('change',stop);return()=>media?.removeEventListener?.('change',stop);},[playing]);
 const choosePhase=i=>{setPhase(i);if(playing)onToggle();};
 return L.jsxs('div',{className:`atlas-tour flow-${chapter.flow} explanation-phase-${phase} ${playing?'guide-playing':'guide-paused'}`,children:[
  L.jsxs('aside',{className:'tour-visual',children:[L.jsx(Dn,{values:chapter.radar}),L.jsx('small',{children:'Havainnollistava hypoteesiprofiili. Muoto ja katkoviiva eivät ole mittausdataa.'})]}),
  chapter.loop&&L.jsx(On,{steps:chapter.loop,kind:chapter.flow}),
  L.jsxs('section',{className:'tour-caption',children:[L.jsx('p',{className:'tour-era',children:`${chapter.era} · Tutkittava tulkinta · ${index+1}/${Tn.length}`}),L.jsx('h2',{children:chapter.title}),L.jsx('p',{className:'guide-point',children:chapter.text}),L.jsxs('div',{className:'guide-phase-copy','aria-live':playing?'off':'polite',children:[L.jsx('strong',{children:phases[phase].title}),L.jsx('p',{children:phases[phase].text})]}),L.jsx('div',{className:'guide-phase-buttons','aria-label':'Selityksen vaiheet',children:phases.map((p,i)=>L.jsx('button',{type:'button',className:i===phase?'active':'','aria-pressed':i===phase,onClick:()=>choosePhase(i),children:`${i+1}. ${p.label||p.title}`},i))}),
   L.jsxs('div',{className:'tour-bar',children:[L.jsx('div',{className:'tour-steps',children:Tn.map((c,i)=>L.jsx('button',{type:'button',className:i===index?'active':'','aria-label':`${c.era}: ${c.title}`,'aria-pressed':i===index,onClick:()=>{if(playing)onToggle();onIndex(i);},children:L.jsx('span',{})},c.era))}),L.jsx('button',{type:'button',onClick:()=>{if(playing)onToggle();onIndex(Math.max(0,index-1));},'aria-label':'Edellinen ajanjakso',children:'←'}),L.jsx('button',{type:'button',onClick:onToggle,'aria-label':playing?'Pysäytä opastus':'Jatka opastusta',children:playing?'Pysäytä':'Jatka'}),L.jsx('button',{type:'button',onClick:()=>{if(phase<phases.length-1)choosePhase(phase+1);else if(index<Tn.length-1){if(playing)onToggle();onIndex(index+1);}else onClose();},'aria-label':'Seuraava selitysvaihe',children:'Seuraava →'}),L.jsx('button',{type:'button',onClick:onClose,'aria-label':'Sulje opastus ja tutki itse',children:'Tutki itse ×'})]}),
   L.jsx('small',{className:'guide-reading-note',children:'Korostettu käyrän osa näyttää tarkasteltavan ajanjakson. Liikkuva virtaus esittää mahdollista yhteyttä; se ei todista syy-seurausta. Voit pysäyttää opastuksen milloin tahansa.'})
  ]})
 ]});
}
