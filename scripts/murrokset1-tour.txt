function kn({index,playing,onIndex,onToggle,onClose}){
 const chapter=Tn[index],phases=chapter.phases||[{title:'Katso ajanjakson asetelmaa',text:chapter.text},{title:'Tutki yhteyksiä',text:'Virtaus esittää tutkittavaa mekanismia.'},{title:'Koettele tulkintaa',text:'Vertaa havaintoja ja vaihtoehtoisia selityksiä.'}];
 const [phase,setPhase]=S.useState(0),[step,setStep]=S.useState(0);
 S.useEffect(()=>{setPhase(0);setStep(0);},[index]);
 S.useEffect(()=>{if(!playing||document.hidden)return;const timer=window.setTimeout(()=>{if(phase===1&&step<3)setStep(s=>s+1);else if(phase<2){setStep(0);setPhase(p=>p+1);}else if(index<Tn.length-1)onIndex(index+1);else onToggle();},phase===1?6500:12000);return()=>window.clearTimeout(timer);},[playing,index,phase,step]);
 S.useEffect(()=>{const media=window.matchMedia?.('(prefers-reduced-motion: reduce)');const stop=()=>{if(media?.matches&&playing)onToggle();};media?.addEventListener?.('change',stop);return()=>media?.removeEventListener?.('change',stop);},[playing]);
 const note=chapter.loop_notes?.[step]||chapter.loop?.[step]||phases[1].text;
 return L.jsxs('aside',{className:`atlas-tour local-guide flow-${chapter.flow} explanation-phase-${phase} loop-step-${step} ${playing?'guide-playing':'guide-paused'}`,'aria-label':'Kuvion vaiheittainen opastaja',children:[
  L.jsxs('header',{className:'local-guide-heading',children:[L.jsx('small',{children:`${chapter.era} · Tutkittava tulkinta`}),L.jsx('h2',{children:chapter.title}),L.jsx('span',{className:'guide-progress',children:`Ajanjakso ${index+1}/${Tn.length} · vaihe ${phase+1}/3`})]}),
  L.jsxs('section',{className:'local-explanation','aria-live':playing?'off':'polite',children:[
   phase===0&&L.jsx('p',{className:'curve-pointer',children:'← Seuraa korostettua käyrän osaa ja sen ajanjaksopainikkeita.'}),
   phase===1&&chapter.loop&&L.jsx(On,{steps:chapter.loop,kind:chapter.flow}),
   L.jsx('h3',{children:phases[phase].title}),L.jsx('p',{children:phases[phase].text}),
   phase===1&&L.jsxs('div',{className:'flow-explanation',children:[L.jsx('strong',{children:`${step+1}. ${chapter.loop?.[step]||'Tutkittava yhteys'}`}),L.jsx('p',{children:note})]}),
   phase===2&&L.jsx('p',{className:'guide-evidence-note',children:'Avaa käyrän ankkuripainike ja tarkista havainto, lähde sekä vaihtoehtoinen selitys. Viivan muoto ei yksin vahvista tulkintaa.'})
  ]}),
  L.jsxs('footer',{className:'local-guide-controls',children:[L.jsx('button',{type:'button',onClick:onToggle,'aria-label':playing?'Pysäytä opastus':'Jatka opastusta',children:playing?'Pysäytä':'Jatka'}),L.jsx('button',{type:'button',onClick:onClose,'aria-label':'Sulje opastus ja tutki itse',children:'Sulje opastus'})]})
 ]});
}
